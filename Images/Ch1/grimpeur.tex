% ---------------------------------------------------------------
%  grimpeur.tex --- bonhomme bâton en train d'escalader (TikZ)
%
%  Usage :  \pic[<options>] at (x,y) {grimpeur};
%           (le point d'ancrage est le bassin du bonhomme)
%
%  Options :  couleur grimpeur=<couleur>   défaut : black
%             halo grimpeur=<couleur>      défaut : white
%             avec halo=true|false         défaut : true
%             epaisseur grimpeur=<dim>     défaut : 0.8pt
%             sac=true|false               défaut : true  (sac à magnésie)
%             scale=<h>   ->  hauteur totale du bonhomme ~= <h> cm
%             xscale=-<h> ->  grimpeur tourné vers la gauche
% ---------------------------------------------------------------
\usetikzlibrary{calc}

\newif\ifGrimpeurHalo
\colorlet{GrimpeurCouleur}{black}
\colorlet{GrimpeurHaloCouleur}{white}
\newcommand{\GrimpeurEp}{0.8pt}

\tikzset{
  couleur grimpeur/.code   = {\colorlet{GrimpeurCouleur}{#1}},
  halo grimpeur/.code      = {\colorlet{GrimpeurHaloCouleur}{#1}},
  avec halo/.is if         = GrimpeurHalo,
  sac/.is if               = GrimpeurSac,
  epaisseur grimpeur/.code = {\renewcommand{\GrimpeurEp}{#1}},
  avec halo                = true,
}

% #1 = style de trait, #2 = couleur de remplissage
\newcommand{\GrimpeurTracer}[2]{%
  % tronc : bassin -> épaules
  \draw[#1] (0,0) -- (0.03,0.25);
  % bras arrière : coude bas, main verrouillée à hauteur de poitrine
  \draw[#1] (0.03,0.25) -- (-0.14,0.16) -- (-0.10,0.33);
  % bras avant : tendu vers la prise du haut
  \draw[#1] (0.03,0.25) -- (0.13,0.36) -- (0.18,0.55);
  % jambe avant : genou sorti, pied haut sur la prise
  \draw[#1] (0,0) -- (0.16,-0.08) -- (0.10,-0.24);
  % jambe arrière : tendue vers le bas
  \draw[#1] (0,0) -- (-0.06,-0.25) -- (-0.01,-0.48);
  % mains et pieds
  \foreach \p in {(0.18,0.55),(-0.10,0.33),(0.10,-0.24),(-0.01,-0.48)}
    {\draw[#1,fill=#2] \p circle[radius=0.020];}
  % tête
  \draw[#1,fill=#2] (0.055,0.36) circle[radius=0.060];
}

\tikzset{
  grimpeur/.pic={
    \ifGrimpeurHalo
      \GrimpeurTracer{line width={\dimexpr\GrimpeurEp+1.4pt\relax},
                      draw=GrimpeurHaloCouleur, line cap=round, line join=round}%
                     {GrimpeurHaloCouleur}
    \fi
    \GrimpeurTracer{line width=\GrimpeurEp, draw=GrimpeurCouleur,
                    line cap=round, line join=round}{GrimpeurCouleur}
  },
}
