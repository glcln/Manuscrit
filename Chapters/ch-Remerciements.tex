\newcommand{\chatcleanup}{%
  \chatagainfalse
  \ifnum\lastnodetype=11
    \ifnum\gluestretchorder\lastskip>0 \chatpartrue\fi
    \chatfoundtrue\unskip\chatagaintrue
  \else\ifnum\lastnodetype=12
    \unkern\chatagaintrue
  \else\ifnum\lastnodetype=13
    \chatfoundtrue\unpenalty\chatagaintrue
  \fi\fi\fi
  \ifchatagain\expandafter\chatcleanup\fi}

\newcommand{\chatpassA}{%
  \global\setbox\chatlinebox=\lastbox

      \ifchatfound\ifchatpar\penalty2 \else\penalty1 \fi\fi
      \box\chatlinebox\unvbox\chataccbox}%
    \expandafter\chatpassA
  \fi}

\newcommand{\chatpassB}{%
  \global\setbox\chatlinebox=\lastbox
  \ifvoid\chatlinebox\else
    \chatabove=\lastpenalty \unpenalty
    \global\setbox\chatoutbox=\vbox{%
      \chatslice
      \ifnum\chatbelow=0 \nointerlineskip\chatbottomcap
      \else\ifnum\chatbelow=2 \penalty0 \nointerlineskip\chatgapslice
      \else
        \ifnum\chatabove=1 \ifnum\chatbelowtwo=1 \penalty0 \else\penalty10000 \fi
        \else\penalty10000 \fi
      \fi\fi
      \unvbox\chatoutbox}%
    \chatbelowtwo=\chatbelow \chatbelow=\chatabove
    \expandafter\chatpassB
  \fi}

\newcommand{\botmsg}[1]{%
  \par\vspace{\chatgap}%
  \begingroup
    \chatfont
    \setlength{\chatbubW}{\dimexpr\chatW-7.5mm\relax}%
    \setbox\chatrawbox=\vbox{\ifdefined\nolinenumbers\nolinenumbers\fi
      \hsize=\chatbotW \parindent=0pt
      \leftskip=0pt \rightskip=0pt \parskip=0pt plus 1fil #1\par}%
    \global\setbox\chataccbox=\vbox{}%
    \setbox\chatscratchbox=\vbox{\unvbox\chatrawbox \chatcleanup \chatpassA}%
    \global\setbox\chatoutbox=\vbox{}%
    \setbox\chatscratchbox=\vbox{\unvbox\chataccbox
      \chatbelow=0 \chatbelowtwo=0 \chatpassB}%
    \chatbotheader
    \nointerlineskip\chattopcap
    \nointerlineskip\unvbox\chatoutbox
  \endgroup
  \par
}
